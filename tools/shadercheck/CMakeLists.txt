# shadercheck: renders libultraship's GLES2 shader template (prism) for many combiner setups, for glslang and lima.
# PRISM_DIR and SPDLOG_DIR: the sources the game's build fetched (build.sh shaders); fetched here when not given.
cmake_minimum_required(VERSION 3.20)
project(shadercheck C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_C_STANDARD 11)
cmake_policy(SET CMP0169 OLD)
if(NOT PRISM_DIR)
    include(FetchContent)
    # the commit libultraship 6b861a64 uses (cmake/dependencies/common.cmake)
    FetchContent_Declare(prism GIT_REPOSITORY https://github.com/KiritoDv/prism-processor.git
        GIT_TAG aa8370981b2cf57c46172e6aa639d720137f9a92)
    FetchContent_Declare(spdlog GIT_REPOSITORY https://github.com/gabime/spdlog.git GIT_TAG v1.16.0)
    FetchContent_Populate(prism)
    FetchContent_Populate(spdlog)
    set(PRISM_DIR ${prism_SOURCE_DIR})
    set(SPDLOG_DIR ${spdlog_SOURCE_DIR})
endif()
file(GLOB_RECURSE PRISM_SRC ${PRISM_DIR}/src/prism/*.cpp ${PRISM_DIR}/src/prism/*.c)
add_executable(shadercheck main.cpp ${PRISM_SRC})
target_include_directories(shadercheck PRIVATE ${PRISM_DIR}/src ${PRISM_DIR}/lib ${SPDLOG_DIR}/include)
target_compile_options(shadercheck PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-Wno-narrowing>)
